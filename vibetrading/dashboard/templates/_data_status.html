{# One status pill. Every market-data number on screen sits next to one of these, so a live price is never mistaken for a simulated or stale one. #} {% macro status_pill(status, extra="") %} {% set label = status.value if status.value is defined else status %} {{ label | replace("_", " ") | upper }} {% endmacro %} {% macro number(value, fmt="%.2f", fallback="—") %} {%- if value is none -%}{{ fallback }}{%- else -%}{{ fmt | format(value) }}{%- endif -%} {% endmacro %} {% macro integer(value, fallback="—") %} {%- if value is none -%}{{ fallback }}{%- else -%}{{ "{:,}".format(value) }}{%- endif -%} {% endmacro %} {% macro signed(value, fmt="%+.2f", fallback="—") %} {%- if value is none -%} {{ fallback }} {%- else -%} {{ fmt | format(value) }} {%- endif -%} {% endmacro %}
